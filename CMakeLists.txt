cmake_minimum_required(VERSION 3.24)
project(libcfindata VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
option(FG_SANITIZE "Enable ASan (and fatal UBSan with GCC/Clang)" OFF)

add_library(cfindata STATIC src/fg_core.c src/fg_data.c src/fg_transport.c
    src/fg_sec.c src/fg_html.c src/fg_xbrl.c src/fg_sources_internal.c
    src/fg_fred.c src/fg_positioning.c)
target_include_directories(cfindata PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(cfindata PUBLIC /W4)
    target_compile_definitions(cfindata PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(cfindata PUBLIC -Wall -Wextra -Wpedantic)
    target_link_libraries(cfindata PUBLIC m)
endif()
if(WIN32)
    target_link_libraries(cfindata PRIVATE winhttp)
else()
    find_package(CURL REQUIRED)
    target_link_libraries(cfindata PRIVATE CURL::libcurl)
endif()
if(FG_SANITIZE)
    if(MSVC)
        string(REGEX REPLACE "/RTC[1su]+" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
        target_compile_options(cfindata PUBLIC /fsanitize=address)
        target_link_options(cfindata PUBLIC /INCREMENTAL:NO)
    else()
        target_compile_options(cfindata PUBLIC -fsanitize=address,undefined
            -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
        target_link_options(cfindata PUBLIC -fsanitize=address,undefined)
    endif()
endif()

add_executable(cfindata_cli src/main.c src/fg_cli.c)
set_target_properties(cfindata_cli PROPERTIES OUTPUT_NAME cfindata)
target_link_libraries(cfindata_cli PRIVATE cfindata)
if(MSVC)
    target_compile_definitions(cfindata_cli PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(MINGW)
    target_link_options(cfindata_cli PRIVATE -municode)
endif()

if(BUILD_TESTING)
    add_executable(test_data tests/test_data.c)
    target_link_libraries(test_data PRIVATE cfindata)
    add_test(NAME data COMMAND test_data "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/core")
    add_executable(test_transport tests/test_transport.c)
    target_link_libraries(test_transport PRIVATE cfindata)
    add_test(NAME transport COMMAND test_transport)
    add_executable(test_sec tests/test_sec.c)
    target_link_libraries(test_sec PRIVATE cfindata)
    add_test(NAME sec COMMAND test_sec "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sec")
    add_executable(test_sources tests/test_sources.c)
    target_link_libraries(test_sources PRIVATE cfindata)
    add_test(NAME sources COMMAND test_sources "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sources")
    add_executable(test_cli tests/test_cli.c src/fg_cli.c)
    target_link_libraries(test_cli PRIVATE cfindata)
    if(MSVC)
        target_compile_definitions(test_cli PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME cli_fixtures COMMAND test_cli "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME cli_help COMMAND cfindata_cli --help)
    set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "fred observations")
    add_test(NAME cli_invalid COMMAND cfindata_cli invalid-command)
    set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
    add_test(NAME cli_unicode COMMAND "${CMAKE_COMMAND}"
        "-DCLI=$<TARGET_FILE:cfindata_cli>" "-DTEST_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}/tmp"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli-unicode.cmake")
    find_program(POWERSHELL_EXECUTABLE NAMES pwsh)
    if(POWERSHELL_EXECUTABLE)
        add_test(NAME transport_loopback COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile
            -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport-loopback.ps1"
            -TestExecutable "$<TARGET_FILE:test_transport>")
        set_tests_properties(transport_loopback PROPERTIES LABELS cfindata TIMEOUT 90
            RESOURCE_LOCK transport_files)
        add_test(NAME dd_adoption COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile
                 -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/dd-adoption.ps1")
        set_tests_properties(dd_adoption PROPERTIES LABELS adoption TIMEOUT 120)
    endif()
    set_tests_properties(transport PROPERTIES RESOURCE_LOCK transport_files)
    set_tests_properties(data transport sec sources cli_fixtures cli_help cli_invalid cli_unicode
                         PROPERTIES LABELS cfindata TIMEOUT 120)
endif()
